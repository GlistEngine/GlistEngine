42struct gGUIBBCodeColor {

56struct gGUIBBCodeStyle {
57 bool bold = false;
58 bool italic = false;
59 bool underline = false;
60 bool strikethrough = false;
61 bool code = false;
62 bool hasColor = false;
63 gGUIBBCodeColor color;
64 bool hasBgColor = false;
65 gGUIBBCodeColor bgColor;
66 int fontSize = 0; // 0 = inherit base font size
67 bool hasUrl = false;
68 std::string url;
69 int linkId = 0;
70
71 bool operator==(const gGUIBBCodeStyle& other) const {
72 return bold == other.bold &&
73 italic == other.italic &&
74 underline == other.underline &&
75 strikethrough == other.strikethrough &&
76 code == other.code &&
77 hasColor == other.hasColor &&
78 (!hasColor || color == other.color) &&
79 hasBgColor == other.hasBgColor &&
80 (!hasBgColor || bgColor == other.bgColor) &&
81 fontSize == other.fontSize &&
82 hasUrl == other.hasUrl &&
83 linkId == other.linkId &&
84 (!hasUrl || url == other.url);
85 }
86 bool operator!=(const gGUIBBCodeStyle& other) const {
87 return !(*this == other);
88 }
89};
