56struct gGUIBBCodeStyle {
57 bool bold = false;
58 bool italic = false;
59 bool underline = false;
60 bool strikethrough = false;
61 bool code = false;
62 bool hasColor = false;
63 gGUIBBCodeColor color;
64 bool hasBgColor = false;
65 gGUIBBCodeColor bgColor;
66 int fontSize = 0; // 0 = inherit textbox base font size
67
68 bool operator==(const gGUIBBCodeStyle& other) const {
69 return bold == other.bold &&
70 italic == other.italic &&
71 underline == other.underline &&
72 strikethrough == other.strikethrough &&
73 code == other.code &&
74 hasColor == other.hasColor &&
75 (!hasColor || color == other.color) &&
76 hasBgColor == other.hasBgColor &&
77 (!hasBgColor || bgColor == other.bgColor) &&
78 fontSize == other.fontSize;
79 }
80 bool operator!=(const gGUIBBCodeStyle& other) const {
81 return !(*this == other);
82 }
83};
