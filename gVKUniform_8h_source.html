1/*
2 * gVKUniform.h
3 *
4 * The per-frame uniform buffer of the Vulkan 3D path: camera matrices and the
5 * scene's lights, in the std140 layout the mesh3d shaders declare.
6 *
7 * There is one buffer per frame in flight rather than a single shared one. The CPU
8 * writes next frame's values while the GPU may still be reading the previous
9 * frame's, so a single buffer would be rewritten underneath work already submitted.
10 * Each buffer stays permanently mapped: the data is rewritten every frame, and
11 * mapping and unmapping around each write would cost more than the memory saved.
12 *
13 * Created by: Veysel Burak Eroglu.
14 */
15

27// How many lights the shader's array holds. Mirrors GLIST_MAX_LIGHTS in
28// gRenderer.h; the static_assert in gVKUniform.cpp keeps the two from drifting.
29inline constexpr int GVK_MAX_LIGHTS = 8;

31/*
32 * One light, laid out to match the Light struct in mesh3d.frag under std140. The
33 * alignas values are what produce that layout: a vec3 aligns to 16 bytes even
34 * though it occupies 12, which is the rule that trips up hand-written UBO structs.
35 *
36 * Deliberately the same shape as gRenderer::gSceneLightData, since the OpenGL path
37 * already feeds that exact layout to its own shader.
38 */
39struct alignas(16) gVKLightData {
40 alignas(4) int type;
41 alignas(16) glm::vec3 position;
42 alignas(16) glm::vec3 direction;
43 alignas(16) glm::vec4 ambient;
44 alignas(16) glm::vec4 diffuse;
45 alignas(16) glm::vec4 specular;
46 alignas(4) float constant;
47 alignas(4) float linear;
48 alignas(4) float quadratic;
49 alignas(4) float cutoff;
50 alignas(4) float outercutoff;
51};
52
53/*
54 * The whole Scene uniform block, matching set 0 binding 0 of the mesh3d shaders.
55 * pad0/pad1 are not decoration: std140 starts the light array on a 16 byte
56 * boundary, so the two ints before it have to be followed by eight bytes of
57 * nothing, and naming them keeps that visible instead of accidental.
58 */
59struct alignas(16) gVKSceneUniforms {
60 alignas(16) glm::mat4 projection;
61 alignas(16) glm::mat4 view;
62 // World space -> the light's clip space, for looking a fragment up in the shadow
63 // map. Identity when there is no shadow map, in which case useshadowmap is 0 and
64 // the shader never uses it.
65 alignas(16) glm::mat4 lightmatrix;
66 alignas(16) glm::vec4 viewpos;
67 alignas(16) glm::vec4 globalambientcolor;
68 // The renderer's current draw colour, kept for layout only: both 3D shaders
69 // still declare this field, so the struct has to match, but neither reads it.
70 //
71 // It started here and had to move. color_frag.glsl ends with
72 // "result * renderColor", and OpenGL republishes the value inside setColor, so a
73 // canvas that recolours between two meshes gets a different colour on each. A
74 // uniform block cannot follow that: it is read when the commands execute, not
75 // when they are recorded, so every draw of the frame sees whichever value was
76 // written last. drawMesh3D folds the colour into the material push constant
77 // instead, which is per draw by construction.
78 alignas(16) glm::vec4 rendercolor;
79 // xyz is the shadow-casting light's position, which the depth bias is computed
80 // from; w is 0 or 1 for whether a shadow map is bound at all.
81 alignas(16) glm::vec4 shadowlightpos;
82 alignas(4) int lightnum;
83 alignas(4) int enabledlights;
84 // 5x5 PCF instead of 3x3, mirroring ENABLE_SOFT_SHADOWS on the OpenGL side.
85 alignas(4) int softshadows;
86 alignas(4) int pad0;
87 gVKLightData lights[GVK_MAX_LIGHTS];
88};

90/*
91 * The push constant block of the mesh3d pipeline. Held to 128 bytes, the smallest
92 * range Vulkan guarantees, so no implementation has to be asked for more.
93 */
94struct gVKMeshPush {
95 glm::mat4 model;
96 glm::vec4 ambient;
97 glm::vec4 diffuse;
98 glm::vec4 specular;
99 // x is shininess; the rest keeps the block vec4 aligned.
100 glm::vec4 misc;
101};
102
103/*
104 * The push constant block of the PBR pipeline, matching mesh3dpbr.*. Smaller than
105 * the one above because a PBR material carries no colours - everything comes from
106 * the maps, or from the shader's defaults where a map is absent.
107 */
108struct gVKPbrPush {
109 glm::mat4 model;
110 // maps0 = albedo, normal, metallic, roughness; maps1.x = ao. Non-zero means the
111 // material supplies that map.
112 glm::ivec4 maps0;
113 glm::ivec4 maps1;
114};
115
116/*
117 * The push constant block of the shadow pipeline, matching shadow3d.*. lightmodel is
118 * lightProjection * lightView * model already multiplied out, so the depth pass needs
119 * no matrices of its own.
120 */
121struct gVKShadowPush {
122 glm::mat4 lightmodel;
123 // x is non-zero when the caster has a cutout diffuse map; the rest keeps the
124 // block vec4 aligned.
125 glm::vec4 misc;
126};

128/*
129 * Creates one uniform buffer and one descriptor set per frame in flight. Must run
