1/*
2 * gVKReflect.h
3 *
4 * Reads a compiled SPIR-V module back and reports what the shader expects from
5 * the pipeline: its vertex inputs, its push constant block and its descriptor
6 * bindings. gVKPipeline uses this instead of repeating those declarations in
7 * C++, so editing a .vert/.frag under graphics/shaders/vk is enough - the
8 * pipeline layout follows the shader on its own.
9 *
10 * This is a deliberately small reflector: it understands the subset of SPIR-V
11 * that glslc emits for the engine's 2D shaders (scalars, vectors, matrices,
12 * arrays, structs, images and samplers) rather than the whole specification.
13 * Anything it does not recognise is skipped with a warning instead of guessed
14 * at, so a shader using an unsupported construct fails loudly at pipeline
15 * creation rather than silently rendering wrong.
16 */

31// One descriptor a shader declares, in the shape vkCreateDescriptorSetLayout
32// wants. stages accumulates while several stages of the same pipeline are
33// reflected, so a sampler read by both the vertex and fragment shader ends up
34// with both bits set.
35struct gVKReflectedBinding {
36 uint32_t set = 0;
37 uint32_t binding = 0;
38 uint32_t count = 1;
39 VkDescriptorType type = VK_DESCRIPTOR_TYPE_COMBINED_IMAGE_SAMPLER;
40 VkShaderStageFlags stages = 0;
41};
42
43// The merged interface of every stage of one pipeline. Reflect the vertex and
44// the fragment module into the same instance to get the whole picture: vertex
45// inputs come from the vertex stage only, while push constants and descriptors
46// are unioned across stages.
47struct gVKReflectedLayout {
48 // Which stages have been reflected into this layout so far.
49 VkShaderStageFlags stages = 0;
50 // Sorted by location, with offsets packed in that order (which is how the
51 // draw path lays its vertex structs out) and stride set to the total.
52 std::vector<VkVertexInputAttributeDescription> vertexattributes;
53 uint32_t vertexstride = 0;
54 // 0 when the pipeline uses no push constants.
55 uint32_t pushconstantsize = 0;
56 VkShaderStageFlags pushconstantstages = 0;
57 // Sorted by (set, binding).
58 std::vector<gVKReflectedBinding> bindings;
59
60 // Highest descriptor set index used, plus one; 0 when there are none.
61 uint32_t getSetCount() const;
62 // The bindings belonging to one set, in binding order.
63 std::vector<VkDescriptorSetLayoutBinding> getSetBindings(uint32_t set) const;
64};
