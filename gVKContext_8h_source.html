99// Multisampling and the render-pass compatibility problem it creates.
100//
101// A graphics pipeline is baked against a render pass, and the sample count of
102// every attachment is part of render-pass *compatibility*: a pipeline built for a
103// 4x pass cannot be recorded into a 1x one, nor the other way round. This backend
104// has three kinds of pass - the screen pass, one pass per offscreen render target
105// (gFbo), and the shadow pass - and the same 2D/3D pipelines are recorded into the
106// first two. Switching the screen pass to 4x therefore breaks every draw into an
107// FBO unless something is done about it.
108//
109// The two honest ways out are to put every pass on the same sample count, or to
110// key the pipelines by sample count and build the variants that are actually used.
111// This backend does the second, for three reasons:
112//
113// - The shadow pass must stay 1x. Multisampling a depth-only map that is compared
114// against rather than displayed buys nothing and costs bandwidth and memory on
115// exactly the mobile parts this backend targets. So "one count everywhere" is
116// already impossible; the only question is where the seam sits.
117// - An FBO's attachments are gTextures the application samples afterwards. Making
118// those passes multisampled means a transient MSAA image *plus* the resolve
119// target for every render target the app owns, and a depth-only FBO - which
120// gFbo produces for depth prepasses - has no resolve path at all before VK 1.2's
121// separate depth/stencil resolve. A post-processing chain wants the raw 1x
122// buffer regardless.
123// - The cost of the second design is one extra set of VkPipeline objects, and only
124// while MSAA is on. They come out of the same VkPipelineCache and reuse the same
125// modules and layouts, so the second build is a fraction of the first.
126//
127// So: index 0 is the single-sample build, used by every offscreen pass and by the
128// screen pass while MSAA is off; index 1 is the build for the screen pass's sample
129// count and only exists while that is above one. getActivePipelineVariant() is what
130// the pipeline getters index with, and whichever pass opens sets it.
131inline constexpr uint32_t GVK_PIPELINE_SAMPLE_VARIANTS = 2;
132
133// The forms one pipeline family takes at a single sample count. Which of them get
134// built is decided per family by gvkPipelineOptions; the ones that are not stay
135// null and the getters fall back to the plain pipeline. gVKContext holds one array
136// of these per family, indexed by sample-count variant.
137struct gVKPipelineVariants {
138 VkPipeline pipeline = VK_NULL_HANDLE;
139 // The same pipeline with a line topology, for stroking unfilled shapes and for
140 // meshes drawn as wireframe.
141 VkPipeline linepipeline = VK_NULL_HANDLE;
142 // The same pipeline with the opposite blend setting; blending cannot be dynamic
143 // state, so following the renderer's setting means choosing a pipeline.
144 VkPipeline blendvariantpipeline = VK_NULL_HANDLE;
145 // The same pipeline adding instead of compositing, for BLENDMODE_ADDITIVE.
146 VkPipeline additivepipeline = VK_NULL_HANDLE;
147 // The same pipeline with the cutout discard specialised out; see
148 // gvkPipelineOptions::nocutoutvariant in gVKPipeline.cpp.
149 VkPipeline nocutoutpipeline = VK_NULL_HANDLE;
150};

584 // Per-frame vertex ring. resetDynamicVertices() rewinds the current frame's
585 // buffer at frame start; pushDynamicVertices() appends vertex bytes (16-byte
586 // aligned) and returns the byte offset to bind from, or VK_WHOLE_SIZE when the
587 // buffer is full.
588 void resetDynamicVertices() {
589 if(!dynvertexoffsets.empty()) dynvertexoffsets[currentframe] = 0;
590 framesequence++;
591 }
592 // Frames since start-up, unlike currentframe which cycles 0..1. A per-frame ring
593 // of any kind has to compare against this: the cycling index repeats, so a user
594 // of the ring that skips a frame would see the same index twice running and
595 // conclude nothing had changed.
596 uint64_t getFrameSequence() const { return framesequence; }
597 VkBuffer getCurrentDynamicVertexBuffer() {
598 return dynvertexbuffers.empty() ? VK_NULL_HANDLE : dynvertexbuffers[currentframe];

600 // alignment is the caller's vertex stride wherever the caller means to reach its
601 // slice through firstVertex rather than by binding at the offset: the index is
602 // offset/stride, so the offset has to be a whole number of vertices from the
603 // start of the buffer. Paths that bind at their own offset can leave it at the
604 // default. Getting this wrong shifts a draw by a fraction of a vertex and
605 // scrambles it, so it is a parameter rather than an assumption - the ring is
606 // shared, and one path's stride is not another's.
607 VkDeviceSize pushDynamicVertices(const void* data, VkDeviceSize size, VkDeviceSize alignment = 16) {
608 if(dynvertexmapped.empty()) return VK_WHOLE_SIZE;
609 if(alignment == 0) alignment = 1;
610 VkDeviceSize offset = ((dynvertexoffsets[currentframe] + alignment - 1) / alignment) * alignment;
611 if(offset + size > dynvertexcapacity) return VK_WHOLE_SIZE;
612 std::memcpy(static_cast<char*>(dynvertexmapped[currentframe]) + offset, data, size);
613 dynvertexoffsets[currentframe] = offset + size;
614 return offset;
615 }

617 // The same idea again, for 3D meshes whose vertices the CPU rewrites: a per
618 // frame arena that every upload takes its own slice of.
619 //
620 // One slice per upload rather than one buffer per mesh, because a mesh can be
621 // uploaded to several times in a single frame and drawn in between. A game
622 // posing one soldier model once per enemy does exactly that, and a single
623 // buffer per mesh gives every one of them the last pose written - the draws are
624 // recorded now and executed later, so they all read whatever the buffer holds
625 // by the end of the frame. Slices keep each draw pointing at the data it was
626 // recorded with.
627 //
628 // pushMeshData returns the byte offset to bind from, or VK_WHOLE_SIZE when the
629 // arena is full - the caller falls back to its own buffer for that frame, and
630 // the arena grows to the high-water mark at the next frame start.
631 void resetMeshArena() {
632 if(mesharenaoffsets.empty()) return;
633 if(mesharenaoffsets[currentframe] > mesharenahighwater) {
634 mesharenahighwater = mesharenaoffsets[currentframe];
635 }
636 mesharenaoffsets[currentframe] = 0;
637 meshgeneration++;
638 }
639 VkBuffer getCurrentMeshArena() {
640 return mesharenabuffers.empty() ? VK_NULL_HANDLE : mesharenabuffers[currentframe];
641 }
642 VkDeviceSize pushMeshData(const void* data, VkDeviceSize size) {
643 if(mesharenamapped.empty() || mesharenamapped[currentframe] == nullptr) return VK_WHOLE_SIZE;
644 VkDeviceSize offset = (mesharenaoffsets[currentframe] + 63) & ~static_cast<VkDeviceSize>(63);
645 if(offset + size > mesharenacapacity) {
646 // Remember what was actually asked for, so the grow at the next reset is
647 // big enough rather than one slice short of it.
648 mesharenahighwater = std::max(mesharenahighwater, offset + size);
649 return VK_WHOLE_SIZE;
650 }
651 std::memcpy(static_cast<char*>(mesharenamapped[currentframe]) + offset, data, size);
652 mesharenaoffsets[currentframe] = offset + size;
653 return offset;
654 }

657 // upload compares this before trusting the slice it kept.

736 // Vertex and index bindings are command buffer state in their own right: binding a
737 // pipeline does not disturb them, so a run of draws over one mesh - a crowd posed
738 // from the same model, or the same buffer drawn once per material - can rebind the
739 // same handles hundreds of times per frame for nothing. The contents behind a
740 // handle may change between draws; that does not matter here, because a draw reads
741 // the buffer when the GPU executes it, not when the bind was recorded.
742 //
743 // Every vkCmdBindVertexBuffers / vkCmdBindIndexBuffer in the backend goes through
744 // these. One that did not would leave the cache describing a binding that is no
745 // longer current, and the next draw would skip a bind it actually needed.
746 bool shouldBindVertexBuffers(const VkBuffer* buffers, const VkDeviceSize* offsets, uint32_t count) {
747 if(count > gvkmaxvertexbindings) return true;
748 if(recordedvertexcount == count
749 && std::equal(buffers, buffers + count, recordedvertexbuffers.begin())
750 && std::equal(offsets, offsets + count, recordedvertexoffsets.begin())) {
751 return false;
752 }
753 recordedvertexcount = count;
754 std::copy(buffers, buffers + count, recordedvertexbuffers.begin());
755 std::copy(offsets, offsets + count, recordedvertexoffsets.begin());
756 return true;
757 }

759 if(recordedindexbuffer == buffer && recordedindexoffset == offset
760 && recordedindextype == type) return false;
761 recordedindexbuffer = buffer;
762 recordedindexoffset = offset;
763 recordedindextype = type;
764 return true;
765 }
