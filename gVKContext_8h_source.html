91// Every Vulkan handle lives here so the header stays Vulkan free, right next to
92// the developer facing settings that shape initialisation. Handles start as
93// VK_NULL_HANDLE, which is what makes the "destroy only if non null" teardown
94// correct even when initialisation fails half way through.
95//
96// The public surface is accessor based: settings go in through setters, and both
97// settings and handles come back out through pointer returning getters. Handing
98// back the address of a handle is deliberate - that is exactly the shape most
99// Vulkan entry points want for their out parameters, so the same getter both
100// reads a handle and receives it when the next phase (swapchain, pipeline, ...)
101// is wired up. gVKRenderEngine drives creation, so it is a friend and reaches
102// the raw members directly; every other consumer goes through the accessors.
103struct gVKContext {
104 friend class gVKRenderEngine;
105 // The backend's own modules create and tear down the members below, so they
106 // reach them directly just like gVKRenderEngine does. Everything outside the
107 // backend goes through the accessors.

401 // Per-frame vertex ring. resetDynamicVertices() rewinds the current frame's
402 // buffer at frame start; pushDynamicVertices() appends vertex bytes (16-byte
403 // aligned) and returns the byte offset to bind from, or VK_WHOLE_SIZE when the
404 // buffer is full.
405 void resetDynamicVertices() {
406 if(!dynvertexoffsets.empty()) dynvertexoffsets[currentframe] = 0;
407 }
408 VkBuffer getCurrentDynamicVertexBuffer() {
409 return dynvertexbuffers.empty() ? VK_NULL_HANDLE : dynvertexbuffers[currentframe];
410 }
411 VkDeviceSize pushDynamicVertices(const void* data, VkDeviceSize size) {
412 if(dynvertexmapped.empty()) return VK_WHOLE_SIZE;
413 VkDeviceSize offset = (dynvertexoffsets[currentframe] + 15) & ~static_cast<VkDeviceSize>(15);
414 if(offset + size > dynvertexcapacity) return VK_WHOLE_SIZE;
415 std::memcpy(static_cast<char*>(dynvertexmapped[currentframe]) + offset, data, size);
416 dynvertexoffsets[currentframe] = offset + size;
417 return offset;
418 }
