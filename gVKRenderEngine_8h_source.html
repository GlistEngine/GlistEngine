21// What a vertex array id stands for: the pair of buffer ids that were bound while
22// it was current, 0 meaning nothing was bound. Vulkan has no vertex array object,
23// but gVbo binds one around every upload and every draw, so remembering the pairing
24// is what lets a draw find the buffers it is about. Defined here rather than in
25// gVKMeshBuffer.h because it holds no Vulkan types and so needs no Vulkan headers.
26struct gVKVertexArray {
27 GLuint vertexbuffer = 0;
28 GLuint indexbuffer = 0;
29 // A vertex array that is drawn instanced has a second array buffer holding one
30 // model matrix per instance. gVbo uploads it through the same entry points as
31 // the vertex data, so the backend tells them apart by order: the first array
32 // buffer a vertex array sees is its vertices, a later different one is the
33 // instance data. setVertexAttribDivisor confirms it.
34 GLuint instancebuffer = 0;
35};

245 /* ---------------- Vulkan context ---------------- */
246 // gVKContext is opaque here (its layout lives in the .cpp), so a developer can
247 // reach the accessor rich context without this header ever pulling in
248 // <vulkan/vulkan.h>. getContext() creates the context on first access when it
249 // does not exist yet, so on a Vulkan-capable build it never returns null:
250 // getContext()->setMinApiVersion(...) before init() is safe, and init() adopts
251 // this same context. Whether the handles are actually populated is a separate
252 // question - check isInitialized() before using them. (A build without Vulkan
253 // support has no context type, so there it returns null.)
