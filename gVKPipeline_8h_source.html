1/*
2 * gVKPipeline.h
3 *
4 * Builds the graphics pipelines the Vulkan 2D draw path uses: one for coloured
5 * geometry (triangle, rectangle) and one for textured quads (gImage). Also owns
6 * the descriptor-set layouts and the descriptor pool textures draw from.
7 *
8 * Nothing about those layouts is written out here: the vertex attributes, the
9 * push constant range and the descriptor bindings are all reflected out of the
10 * compiled SPIR-V by gVKReflect, so the shader sources under
11 * graphics/shaders are the only place the interface is declared. Editing a
12 * shader is enough - the pipeline follows.
13 *
14 * The entry points gvkCreateGraphicsPipelines / gvkReloadGraphicsPipelines /
15 * gvkDestroyGraphicsPipelines are declared in gVKContext.h (so the context can
16 * befriend them); this header adds only what needs no access to the context.
17 */

37/*
38 * A pipeline built from a shader the application supplied rather than one of the
39 * engine's own. Everything in it is reflected out of the given SPIR-V, exactly as
40 * the built-in pipelines are, so a gShader loaded at run time gets the same
41 * treatment as mesh3d: the shader declares its interface and the layout follows.
42 *
43 * The vertex input is the one exception. It is taken from the engine's own mesh
44 * layout rather than from reflection, because the buffer the draw path binds is
45 * a gVertex array whatever the shader happens to declare. Vulkan allows a
46 * pipeline to describe attributes the shader never reads, so a shader consuming
47 * only position and texcoords still works against the full layout - what it must
48 * not do is declare an input the layout has no attribute for, which
49 * gvkBuildUserPipeline rejects.
50 */
51struct gVKUserPipeline {
52 gVKPipelineVariants variants[GVK_PIPELINE_SAMPLE_VARIANTS];
53 VkPipelineLayout layout = VK_NULL_HANDLE;
54 std::vector<VkDescriptorSetLayout> setlayouts;
55 uint32_t pushsize = 0;
56 VkShaderStageFlags pushstages = 0;
57 // What the shader declared, kept so values can be addressed by name.
58 gVKReflectedLayout reflected;
59 // True when the pipeline describes the engine's gVertex layout, so it can be
60 // drawn with mesh buffers. False means the layout came from reflection and the
61 // shader is fed by the fullscreen quad path instead.
62 bool usesmeshlayout = false;
63 // Bytes per vertex of whichever layout was chosen.
64 uint32_t vertexstride = 0;
65};

67// Builds one. Returns false with everything released on any failure, so a caller

69bool gvkBuildUserPipeline(gVKContext& ctx, const std::vector<uint32_t>& vertSpirv,
70 const std::vector<uint32_t>& fragSpirv, const char* debugName, gVKUserPipeline& out);

72// Releases one. Safe on an all-null instance, and leaves it that way.
73void gvkDestroyUserPipeline(gVKContext& ctx, gVKUserPipeline& pipeline);
